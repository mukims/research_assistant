# Minimal dependencies to run the test suite.
#
#   python -m pip install -e .
#   python -m pip install -r requirements-test.txt
#   CITATION_LOG_FILE=0 python -m pytest tests/ -v
#
# The suite exercises pure logic — citation parsing, paper naming, extractor
# filing, ingestion bookkeeping, retrieval ranking. The heavy stack (ChromaDB,
# PyTorch, the LangChain embedding chain) is imported
# lazily inside the functions the tests never call, so none of it is needed
# here. That is deliberate: it keeps the tests runnable anywhere.
#
# Most tests need no provider SDK: every agent calls
# research_assistant.shared.llm.chat()/chat_stream(), and test_llm.py exercises
# the ollama/openai backends by injecting fakes into sys.modules. A few
# (test_challenger_m2_p1_stress.py, test_p1_remediation.py) patch the real
# ollama.chat / openai client / grobid_client.GrobidClient, so those three small
# pure-Python SDKs are pinned below, as in requirements.txt. chromadb is still not
# needed: tests that reach the store use LanceDB or a fake.
#
# pytest replaces unittest discover — it collects the existing
# unittest.TestCase classes unchanged, so no test file needed to change.
#
# watchdog is here for CI's import sweep, not for the tests themselves:
# agent6_manual_ingestor.py imports it at module scope (unlike the ML stack,
# it is not deferred into a function), so anything that imports that module
# needs it installed. It is pure Python with no compiled-wheel risk, so it
# doesn't compromise this file's "light packages only" invariant.
#
# beautifulsoup4 and lxml back test_extractor.py's TestTeiParsing /
# TestDoiConfidence: agent1_extractor.parse_reference() and
# parse_article_metadata() are real TEI parsing, not opaque logic, and both
# packages are pure-Python-wheel/pinned in requirements.txt already — there
# was never a reason they were untestable here. Pinned to the same versions
# requirements.txt uses, so the tests run against what ships.
beautifulsoup4==4.15.0
lxml==6.1.3
#
# langgraph and streamlit are here for CI's import sweep only, which now
# covers the three root entry points (app.py, orchestrate.py, watch.py) —
# C1 (app.py importing orchestrate as though it were a package module) lived
# in exactly the gap this closes. Both are pure-Python wheels, same versions
# pinned in requirements.txt.
langgraph==1.2.15
streamlit==1.65.0
#
# PyMuPDF backs the task planner's PDF extraction (tasks/extractor.py) and the
# PDF-structure tests in test_task_extractor.py and test_discoverer.py. It is a
# binary wheel but not part of the heavy stack above; same pin as requirements.txt.
PyMuPDF==1.28.2
#
# lancedb and pyarrow: LanceDB is the app's default store (config.USE_LANCEDB), so the
# suite must run against it -- tests/test_lancedb_store.py, test_lance_ingestion.py and
# the challenger suites skip or fail without them. Binary wheels; same pins as requirements.txt.
lancedb==0.41.0
pyarrow==25.0.1
ollama==0.6.3
openai==3.31.0
grobid-client-python==0.2.0
pytest~=9.1
numpy~=2.2.6
requests~=2.32
watchdog~=6.0
# snowballstemmer backs shared/tokenize.py, which shared/search.py imports at
# module scope; pysbd backs shared/chunking.py. Both are pure Python.
snowballstemmer~=3.1
pysbd~=0.3
# config.py reads .env (and DATA_DIR/.env) through it; without it the app
# silently ignores the backend settings on the data disk.
python-dotenv>=1.0
